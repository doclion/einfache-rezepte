{{- /* Strukturierte Daten. Wichtigste Regeln: - "Recipe" NUR auf Rezeptseiten, die tatsaechlich Zutaten und Zubereitung im Frontmatter haben. "pfad" .Params.image) -}} {{- $r := dict "@context" "https://schema.org" "@type" "Recipe" "name" .Title "description" (partial "beschreibung.html" .) "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "inLanguage" "de-DE" "datePublished" (.Date.Format "2006-01-02") "dateModified" (.Lastmod.Format "2006-01-02") "author" (dict "@type" "Person" "name" $autor) "publisher" $verlag "recipeIngredient" $zutaten "recipeInstructions" $schritte -}} {{- with $bild -}}{{- $r = merge $r (dict "image" (slice .)) -}}{{- end -}} {{- if gt $prep 0 -}}{{- $r = merge $r (dict "prepTime" (printf "PT%dM" $prep)) -}}{{- end -}} {{- if gt $cook 0 -}}{{- $r = merge $r (dict "cookTime" (printf "PT%dM" $cook)) -}}{{- end -}} {{- if gt $gesamt 0 -}}{{- $r = merge $r (dict "totalTime" (printf "PT%dM" $gesamt)) -}}{{- end -}} {{- with .Params.servings -}}{{- $r = merge $r (dict "recipeYield" (printf "%v Portionen" .)) -}}{{- end -}} {{- with .Params.categories -}}{{- $r = merge $r (dict "recipeCategory" (index . 0)) -}}{{- end -}} {{- with .Params.cuisine -}}{{- $r = merge $r (dict "recipeCuisine" .) -}}{{- end -}} {{- with .Params.tags -}}{{- $r = merge $r (dict "keywords" (delimit . ", ")) -}}{{- end -}} {{- if .Params.calories -}} {{- $n := dict "@type" "NutritionInformation" "servingSize" "1 Portion" "calories" (printf "%v kcal" .Params.calories) -}} {{- with .Params.protein -}}{{- $n = merge $n (dict "proteinContent" (printf "%v g" .)) -}}{{- end -}} {{- with .Params.fat -}}{{- $n = merge $n (dict "fatContent" (printf "%v g" .)) -}}{{- end -}} {{- with .Params.carbs -}}{{- $n = merge $n (dict "carbohydrateContent" (printf "%v g" .)) -}}{{- end -}} {{- $r = merge $r (dict "nutrition" $n) -}} {{- end -}} {{- else if eq .Kind "page" -}} {{- $wp := dict "@context" "https://schema.org" "@type" "WebPage" "name" .Title "url" .Permalink "inLanguage" "de-DE" "description" (partial "beschreibung.html" .) "isPartOf" (dict "@type" "WebSite" "name" $s.Title "url" $s.BaseURL) -}} {{- end -}} {{- /* Breadcrumbs auf Rezeptseiten */ -}} {{- if eq .Section "posts" -}} {{- $items := slice (dict "@type" "ListItem" "position" 1 "name" "Start" "item" $s.BaseURL) (dict "@type" "ListItem" "position" 2 "name" "Rezepte" "item" ("posts/" | absURL)) (dict "@type" "ListItem" "position" 3 "name" .Title "item" .Permalink) -}} {{- end -}}
